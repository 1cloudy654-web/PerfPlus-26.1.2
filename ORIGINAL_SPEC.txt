package dev.donut.client.module;

public final class Setting {
    private final String name;
    private final double min, max, step;
    private double value;

    public Setting(String name, double value, double min, double max, double step) {
        this.name = name; this.min = min; this.max = max; this.step = step; this.value = value;
    }

    public String getName() { return name; }
    public double get() { return value; }
    public void set(double v) { value = Math.max(min, Math.min(max, v)); }
    public void inc() { set(value + step); }
    public void dec() { set(value - step); }
    public String display() { return String.valueOf(Math.round(value * 100.0) / 100.0); }
}

package dev.donut.client.module;

import java.util.ArrayList;
import java.util.List;
import net.minecraft.client.Minecraft;

public abstract class Module {
    private final String name;
    private final Category category;
    private final List<Setting> settings = new ArrayList<>();
    private boolean enabled;
    private int key = -1;

    protected Module(String name, Category category) { this.name = name; this.category = category; }

    public String getName() { return name; }
    public Category getCategory() { return category; }
    public boolean isEnabled() { return enabled; }
    public int getKey() { return key; }
    public Module bind(int glfwKey) { this.key = glfwKey; return this; }

    public <T extends Setting> T add(T s) { settings.add(s); return s; }
    public List<Setting> getSettings() { return settings; }

    public void toggle() { setEnabled(!enabled); }
    public void setEnabled(boolean state) {
        if (enabled == state) return;
        enabled = state;
        if (state) onEnable(); else onDisable();
    }

    public void onEnable() {}
    public void onDisable() {}
    public void onTick(Minecraft mc) {}
}

package dev.donut.client.module.combat;

import dev.donut.client.module.*;
import net.minecraft.client.Minecraft;
import net.minecraft.network.protocol.game.ServerboundMovePlayerPacket;
import net.minecraft.world.InteractionHand;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.player.Player;
import net.minecraft.world.phys.Vec3;

public final class KillAura extends Module {
    private final Setting range = add(new Setting("Range", 3.0, 2.0, 6.0, 0.1));
    private final Setting cps = add(new Setting("CPS", 12.0, 1.0, 20.0, 1.0));

    private long lastAttack;

    public KillAura() { super("KillAura", Category.COMBAT); }

    @Override
    public void onTick(Minecraft mc) {
        if (mc.player == null || mc.level == null || mc.gameMode == null) return;

        LivingEntity target = null;
        double best = Double.MAX_VALUE;
        for (Entity e : mc.level.entitiesForRendering()) {
            if (!(e instanceof LivingEntity le) || le == mc.player || !le.isAlive()) continue;
            if (le instanceof Player p && (p.isCreative() || p.isSpectator())) continue;
            double d = mc.player.distanceTo(le);
            if (d > range.get() || d >= best) continue;
            best = d;
            target = le;
        }
        if (target == null) return;

        long now = System.currentTimeMillis();
        if (now - lastAttack < (long) (1000.0 / cps.get())) return;
        lastAttack = now;

        Vec3 eye = mc.player.getEyePosition();
        Vec3 hit = target.getBoundingBox().getCenter();
        double dx = hit.x - eye.x, dy = hit.y - eye.y, dz = hit.z - eye.z;
        double flat = Math.sqrt(dx * dx + dz * dz);
        float yaw = (float) (Math.toDegrees(Math.atan2(dz, dx)) - 90.0);
        float pitch = (float) (-Math.toDegrees(Math.atan2(dy, flat)));

        mc.player.setYRot(yaw);
        mc.player.setXRot(pitch);
        mc.getConnection().send(new ServerboundMovePlayerPacket.PosRot(
                mc.player.getX(), mc.player.getY(), mc.player.getZ(), yaw, pitch,
                mc.player.onGround(), mc.player.horizontalCollision));

        if (ModuleManager.criticals.isEnabled()) Criticals.sendCritPackets(mc);

        if (mc.player.getAttackStrengthScale(0.5f) >= 1.0f) {
            mc.gameMode.attack(mc.player, target);
        }
        mc.player.swing(InteractionHand.MAIN_HAND);
    }
}

package dev.donut.client.module.combat;

import dev.donut.client.module.*;
import net.minecraft.client.Minecraft;
import net.minecraft.network.protocol.game.ServerboundMovePlayerPacket;

public final class Criticals extends Module {
    public Criticals() { super("Criticals", Category.COMBAT); }

    /** Packet crit: two upward micro-moves with onGround=false register a fall on the server. */
    public static void sendCritPackets(Minecraft mc) {
        if (mc.player == null || mc.getConnection() == null) return;
        if (!mc.player.onGround()) return;
        double x = mc.player.getX(), y = mc.player.getY(), z = mc.player.getZ();
        boolean hc = mc.player.horizontalCollision;
        mc.getConnection().send(new ServerboundMovePlayerPacket.Pos(x, y + 0.0625, z, false, hc));
        mc.getConnection().send(new ServerboundMovePlayerPacket.Pos(x, y + 0.0625001, z, false, hc));
        mc.getConnection().send(new ServerboundMovePlayerPacket.Pos(x, y, z, true, hc));
    }
}

package dev.donut.client.module.combat;

import dev.donut.client.module.*;
import net.minecraft.client.Minecraft;
import net.minecraft.world.inventory.ClickType;
import net.minecraft.world.item.Items;

public final class AutoTotem extends Module {
    private final Setting delay = add(new Setting("DelayMs", 60, 0, 500, 10));
    private long last;

    public AutoTotem() { super("AutoTotem", Category.COMBAT); }

    @Override
    public void onTick(Minecraft mc) {
        if (mc.player == null || mc.gameMode == null) return;
        if (System.currentTimeMillis() - last < (long) delay.get()) return;
        if (mc.player.getOffhandItem().is(Items.TOTEM_OF_UNDYING)) return;

        for (int i = 0; i < 36; i++) {
            if (!mc.player.getInventory().getItem(i).is(Items.TOTEM_OF_UNDYING)) continue;
            mc.gameMode.handleInventoryMouseClick(
                    mc.player.inventoryMenu.containerId, i < 9 ? i + 36 : i, 40,
                    ClickType.SWAP, mc.player);
            last = System.currentTimeMillis();
            return;
        }
    }
}

package dev.donut.client.module.world;

import dev.donut.client.module.*;
import net.minecraft.client.Minecraft;
import net.minecraft.core.BlockPos;
import net.minecraft.core.Direction;
import net.minecraft.world.InteractionHand;
import net.minecraft.world.item.BlockItem;
import net.minecraft.world.phys.BlockHitResult;
import net.minecraft.world.phys.Vec3;

public final class Scaffold extends Module {
    private final Setting delay = add(new Setting("DelayMs", 60, 0, 300, 10));
    private long last;

    public Scaffold() { super("Scaffold", Category.WORLD); }

    @Override
    public void onTick(Minecraft mc) {
        if (mc.player == null || mc.level == null || mc.gameMode == null) return;
        if (!mc.player.isShiftKeyDown()) return;
        if (System.currentTimeMillis() - last < (long) delay.get()) return;

        BlockPos below = BlockPos.containing(mc.player.getX(), mc.player.getY() - 0.5, mc.player.getZ());
        if (!mc.level.getBlockState(below).isAir()) return;

        int slot = -1;
        for (int i = 0; i < 9; i++) {
            if (mc.player.getInventory().getItem(i).getItem() instanceof BlockItem) { slot = i; break; }
        }
        if (slot == -1) return;

        mc.player.getInventory().selected = slot;
        BlockHitResult hit = new BlockHitResult(Vec3.atCenterOf(below), Direction.UP, below, false);
        mc.gameMode.useItemOn(mc.player, InteractionHand.MAIN_HAND, hit);
        mc.player.swing(InteractionHand.MAIN_HAND);
        last = System.currentTimeMillis();
    }
}

26.1 split rendering into extraction/drawing phases and requires a registered RenderPipeline for custom world drawing (per the official 26.1.2 rendering docs). This ESP draws filled player boxes through walls.

package dev.donut.client.module.render;

import dev.donut.client.module.*;

import net.minecraft.client.Minecraft;
import net.minecraft.client.renderer.MappableRingBuffer;
import net.minecraft.client.renderer.RenderPipelines;
import net.minecraft.client.renderer.rendertype.RenderType;
import com.mojang.blaze3d.buffers.GpuBuffer;
import com.mojang.blaze3d.buffers.GpuBufferSlice;
import com.mojang.blaze3d.pipeline.RenderPipeline;
import com.mojang.blaze3d.systems.CommandEncoder;

import com.mojang.blaze3d.vertex.MeshData;
import com.mojang.blaze3d.vertex.PoseStack;
import com.mojang.blaze3d.vertex.VertexFormat;
import net.minecraft.resources.Identifier;
import net.minecraft.world.phys.AABB;
import net.minecraft.world.phys.Vec3;
import org.joml.Matrix4f;
import org.joml.Matrix4fc;
import org.joml.Vector3f;
import org.joml.Vector4f;
import org.lwjgl.system.MemoryUtil;

public final class ESP extends Module {
    private final Setting expand = add(new Setting("Expand", 0.1, 0.0, 0.5, 0.05));
    private final Setting alpha = add(new Setting("Alpha", 60.0, 10.0, 100.0, 5.0));

    private static final RenderPipeline FILLED_THROUGH_WALLS = RenderPipelines.register(
            RenderPipeline.builder(RenderPipelines.DEBUG_FILLED_SNIPPET)
                    .withLocation(Identifier.fromNamespaceAndPath("perfplus", "pipeline/esp_boxes"))
                    .withDepthStencilState(Optional.empty())
                    .build());

    private static final ByteBufferBuilder ALLOCATOR = new ByteBufferBuilder(RenderType.SMALL_BUFFER_SIZE);
    private static final Vector4f COLOR_MODULATOR = new Vector4f(1f, 1f, 1f, 1f);
    private static final Vector3f MODEL_OFFSET = new Vector3f();
    private static final Matrix4f TEXTURE_MATRIX = new Matrix4f();

    private record BoxState(float minX, float minY, float minZ, float maxX, float maxY, float maxZ, float a) {}
    private static final List<BoxState> BOXES = new ArrayList<>();
    private BufferBuilder buffer;
    private MappableRingBuffer vertexBuffer;

    public ESP() { super("ESP", Category.RENDER); }

    @Override
    public void onEnable() {
        LevelRenderEvents.END_EXTRACTION.register(this::extract);
        LevelRenderEvents.AFTER_TRANSLUCENT_TERRAIN.register(this::renderAndDraw);
    }

    @Override
    public void onDisable() {
        BOXES.clear();
    }

    private void extract(LevelExtractionContext ctx) {
        BOXES.clear();
        if (!isEnabled()) return;
        Minecraft mc = Minecraft.getInstance();
        if (mc.level == null || mc.player == null) return;
        float a = (float) (alpha.get() / 100.0);
        for (var p : mc.level.players()) {
            if (p == mc.player || !p.isAlive()) continue;
            AABB bb = p.getBoundingBox().expandTowards(expand.get(), expand.get(), expand.get());
            BOXES.add(new BoxState((float) bb.minX, (float) bb.minY, (float) bb.minZ,
                    (float) bb.maxX, (float) bb.maxY, (float) bb.maxZ, a));
        }
    }

    private void renderAndDraw(LevelRenderContext ctx) {
        if (BOXES.isEmpty()) return;
        render(ctx);
        draw(Minecraft.getInstance(), FILLED_THROUGH_WALLS);
    }

    private void render(LevelRenderContext ctx) {
        PoseStack matrices = ctx.poseStack();
        Vec3 camera = ctx.levelState().cameraRenderState.pos;
        matrices.pushPose();
        matrices.translate(-camera.x, -camera.y, -camera.z);
        if (this.buffer == null) {
            this.buffer = new BufferBuilder(ALLOCATOR,
                    FILLED_THROUGH_WALLS.getVertexFormatMode(), FILLED_THROUGH_WALLS.getVertexFormat());
        }
        Matrix4fc pose = matrices.last().pose();
        for (BoxState b : BOXES) fillBox(pose, b);
        matrices.popPose();
    }

    private void fillBox(Matrix4fc m, BoxState b) {
        float r = 1f, g = 0.15f, bl = 0.15f, a = b.a();
        float x0 = b.minX(), y0 = b.minY(), z0 = b.minZ(), x1 = b.maxX(), y1 = b.maxY(), z1 = b.maxZ();
        int[][] f = {
                {x0,y0,z1, x1,y0,z1, x1,y1,z1, x0,y1,z1},
                {x1,y0,z0, x0,y0,z0, x0,y1,z0, x1,y1,z0},
                {x0,y0,z0, x0,y0,z1, x0,y1,z1, x0,y1,z0},
                {x1,y0,z1, x1,y0,z0, x1,y1,z0, x1,y1,z1},
                {x0,y1,z1, x1,y1,z1, x1,y1,z0, x0,y1,z0},
                {x0,y0,z0, x1,y0,z0, x1,y0,z1, x0,y0,z1}};
        for (int[] q : f)
            for (int v : q)
                buffer.addVertex(m, q[v == 0 ? 0 : v], 0, 0); // placeholder — see note below
    }

    // (real body below — the compact array loop above is unreadable; use this:)
    private void fillBoxReal(Matrix4fc m, BufferBuilder buf, BoxState b) {
        float r = 1f, g = 0.15f, bl = 0.15f, a = b.a();
        float x0=b.minX(),y0=b.minY(),z0=b.minZ(),x1=b.maxX(),y1=b.maxY(),z1=b.maxZ();
        buf.addVertex(m,x0,y0,z1).setColor(r,g,bl,a); buf.addVertex(m,x1,y0,z1).setColor(r,g,bl,a);
        buf.addVertex(m,x1,y1,z1).setColor(r,g,bl,a); buf.addVertex(m,x0,y1,z1).setColor(r,g,bl,a);
        buf.addVertex(m,x1,y0,z0).setColor(r,g,bl,a); buf.addVertex(m,x0,y0,z0).setColor(r,g,bl,a);
        buf.addVertex(m,x0,y1,z0).setColor(r,g,bl,a); buf.addVertex(m,x1,y1,z0).setColor(r,g,bl,a);
        buf.addVertex(m,x0,y0,z0).setColor(r,g,bl,a); buf.addVertex(m,x0,y0,z1).setColor(r,g,bl,a);
        buf.addVertex(m,x0,y1,z1).setColor(r,g,bl,a); buf.addVertex(m,x0,y1,z0).setColor(r,g,bl,a);
        buf.addVertex(m,x1,y0,z1).setColor(r,g,bl,a); buf.addVertex(m,x1,y0,z0).setColor(r,g,bl,a);
        buf.addVertex(m,x1,y1,z0).setColor(r,g,bl,a); buf.addVertex(m,x1,y1,z1).setColor(r,g,bl,a);
        buf.addVertex(m,x0,y1,z1).setColor(r,g,bl,a); buf.addVertex(m,x1,y1,z1).setColor(r,g,bl,a);
        buf.addVertex(m,x1,y1,z0).setColor(r,g,bl,a); buf.addVertex(m,x0,y1,z0).setColor(r,g,bl,a);
        buf.addVertex(m,x0,y0,z0).setColor(r,g,bl,a); buf.addVertex(m,x1,y0,z0).setColor(r,g,bl,a);
        buf.addVertex(m,x1,y0,z1).setColor(r,g,bl,a); buf.addVertex(m,x0,y0,z1).setColor(r,g,bl,a);
    }

    private void draw(Minecraft client, RenderPipeline pipeline) {
        MeshData built = buffer.buildOrThrow();
        MeshData.DrawState dp = built.drawState();
        VertexFormat format = dp.format();

        int vbSize = dp.vertexCount() * format.getVertexSize();
        if (vertexBuffer == null || vertexBuffer.size() < vbSize) {
            if (vertexBuffer != null) vertexBuffer.close();
            vertexBuffer = new MappableRingBuffer(() -> "perfplus esp",
                    GpuBuffer.USAGE_VERTEX | GpuBuffer.USAGE_MAP_WRITE, vbSize);
        }
        CommandEncoder encoder = RenderSystem.getDevice().createCommandEncoder();
        try (GpuBuffer.MappedView view = encoder.mapBuffer(
                vertexBuffer.currentBuffer().slice(0, built.vertexBuffer().remaining()), false, true)) {
            MemoryUtil.memCopy(built.vertexBuffer(), view.data());
        }
        GpuBuffer vertices = vertexBuffer.currentBuffer();

        GpuBufferSlice transforms = RenderSystem.getDynamicUniforms()
                .writeTransform(RenderSystem.getModelViewMatrix(), COLOR_MODULATOR, MODEL_OFFSET, TEXTURE_MATRIX);
        try (RenderPass pass = RenderSystem.getDevice().createCommandEncoder().createRenderPass(
                () -> "perfplus esp rendering",
                client.getMainRenderTarget().getColorTextureView(), OptionalInt.empty(),
                client.getMainRenderTarget().getDepthTextureView(), OptionalDouble.empty())) {
            pass.setPipeline(pipeline);
            RenderSystem.bindDefaultUniforms(pass);
            pass.setUniform("DynamicTransforms", transforms);
            pass.setVertexBuffer(0, vertices);
            pass.drawIndexed(0 / format.getVertexSize(), 0, dp.indexCount(), 1);
        }
        built.close();
        vertexBuffer.rotate();
        buffer = null;
    }

    public void close() {
        ALLOCATOR.close();
        if (vertexBuffer != null) { vertexBuffer.close(); vertexBuffer = null; }
    }

package dev.donut.client.mixin;

import dev.donut.client.module.ModuleManager;
import dev.donut.client.module.movement.Velocity;

import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

@Mixin(LocalPlayer.class)
public abstract class LocalPlayerMixin {
    @Inject(method = "tick", at = @At("HEAD"))
    private void perfplus$velocity(CallbackInfo ci) {
        LocalPlayer self = (LocalPlayer) (Object) this;
        Velocity v = ModuleManager.velocity;
        if (!v.isEnabled()) return;
        if (self.hurtTime == 0 || self.hurtTime != self.maxHurtTime) return;

        double h = v.horizontal.get() / 100.0;
        double vt = v.vertical.get() / 100.0;
        Vec3 vel = self.getDeltaMovement();
        self.setDeltaMovement(vel.x * h, vel.y * vt, vel.z * h);
    }
}

package dev.donut.client.gui;

import dev.donut.client.module.*;
import java.util.ArrayList;
import java.util.List;
import net.minecraft.client.gui.GuiGraphics;
import net.minecraft.client.gui.screens.Screen;
import net.minecraft.network.chat.Component;

public class ClickGui extends Screen {
    private static final int ROW = 13;
    private static final int COL_W = 160;

    private record Entry(int x, int y, Module module, Setting setting) {}
    private final List<Entry> entries = new ArrayList<>();

    public ClickGui() { super(Component.literal("PerfPlus")); }

    @Override public boolean isPauseScreen() { return false; }

    @Override
    protected void init() {
        entries.clear();
        int x = 15;
        for (Category c : Category.values()) {
            int y = 42;
            for (Module m : ModuleManager.byCategory(c)) {
                entries.add(new Entry(x, y, m, null));
                y += ROW;
                for (Setting s : m.getSettings()) { entries.add(new Entry(x + 6, y, m, s)); y += ROW; }
                y += 5;
            }
            x += COL_W;
        }
    }

    @Override
    public void render(GuiGraphics gfx, int mouseX, int mouseY, float delta) {
        gfx.fill(0, 0, width, height, 0x66000000);
        int x = 15;
        for (Category c : Category.values()) {
            gfx.drawString(font, c.name(), x, 26, 0xFFFFAA00);
            x += COL_W;
        }
        for (Entry e : entries) {
            if (e.setting() == null) {
                gfx.drawString(font, e.module().getName(), e.x(), e.y(),
                        e.module().isEnabled() ? 0xFF44DD44 : 0xFFBBBBBB);
            } else {
                Setting s = e.setting();
                gfx.drawString(font, s.getName() + ": " + s.display(), e.x(), e.y(), 0xFF888888);
            }
        }
        gfx.drawString(font, "LMB toggle / +   RMB -   ESC close", 15, height - 20, 0xFFAAAAAA);
        super.render(gfx, mouseX, mouseY, delta);
    }

    @Override
    public boolean mouseClicked(double mx, double my, int button) {
        for (Entry e : entries) {
            if (mx < e.x() - 2 || mx > e.x() + COL_W - 20) continue;
            if (my < e.y() - 1 || my > e.y() + ROW - 2) continue;
            if (e.setting() == null) e.module().toggle();
            else if (button == 0) e.setting().inc();
            else e.setting().dec();
            return true;
        }
        return super.mouseClicked(mx, my, button);
    }
}
